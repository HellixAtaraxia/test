---------------------------------------------------------------- Util umum
local function getPath(inst)
	local parts = {}
	local cur = inst
	while cur and cur ~= game do
		local n = cur.Name
		if n:match("^[%a_][%w_]*$") then
			table.insert(parts, 1, "." .. n)
		else
			table.insert(parts, 1, '["' .. n .. '"]')
		end
		cur = cur.Parent
	end
	return "game" .. table.concat(parts)
end

local function vecString(v)
	return string.format("Vector3.new(%.2f, %.2f, %.2f)", v.X, v.Y, v.Z)
end

local function cfString(cf)
	local p = cf.Position
	local rx, ry, rz = cf:ToOrientation()
	return string.format(
		"CFrame.new(%.2f, %.2f, %.2f) * CFrame.Angles(math.rad(%.1f), math.rad(%.1f), math.rad(%.1f))",
		p.X, p.Y, p.Z, math.deg(rx), math.deg(ry), math.deg(rz)
	)
end

local function describe(inst)
	local lines = { "Nama: " .. inst.Name, "Class: " .. inst.ClassName }
	if inst:IsA("BasePart") then
		local p, s = inst.Position, inst.Size
		table.insert(lines, string.format("Pos: %.1f, %.1f, %.1f", p.X, p.Y, p.Z))
		table.insert(lines, string.format("Size: %.1f, %.1f, %.1f", s.X, s.Y, s.Z))
		table.insert(lines, "Material: " .. inst.Material.Name)
		table.insert(lines, "Anchored: " .. tostring(inst.Anchored) .. " | CanCollide: " .. tostring(inst.CanCollide))
	elseif inst:IsA("Model") then
		local p = inst:GetPivot().Position
		table.insert(lines, string.format("Pivot: %.1f, %.1f, %.1f", p.X, p.Y, p.Z))
		table.insert(lines, "Children: " .. #inst:GetChildren())
	else
		table.insert(lines, "Children: " .. #inst:GetChildren())
	end
	table.insert(lines, "Path: " .. getPath(inst))
	return table.concat(lines, "\n")
end

local function copy(text, what)
	if not text or text == "" then
		notify("tidak ada yang disalin")
		return
	end
	print("[DevHelper] " .. text)
	if setclipboard then
		pcall(setclipboard, text)
		notify("disalin: " .. (what or "teks"))
	else
		notify("clipboard tdk tersedia, lihat Output")
	end
end

local function refreshExplorer()

	if not selected then
		pathLabel.Text = " (belum ada yang dipilih)"
		return
	end
	pathLabel.Text = " " .. selected.ClassName .. "\n " .. getPath(selected)
	local kids = selected:GetChildren()
	table.sort(kids, function(a, b) return a.Name:lower() < b.Name:lower() end)
	for i, k in ipairs(kids) do
		if i > 300 then
			listItem(exList, "... dipotong di 300 item", function() end, i)
			break
		end
		listItem(exList, "[" .. k.ClassName .. "] " .. k.Name, function() selectInst(k) end, i)
	end

local touchConns = {}
local debounceObj = nil

local function hookCharacter(char)
	for _, c in ipairs(touchConns) do c:Disconnect() end
	touchConns = {}
	for _, part in ipairs(char:GetDescendants()) do
		if part:IsA("BasePart") then
			table.insert(touchConns, part.Touched:Connect(function(other)
				if other:IsDescendantOf(char) or other == debounceObj then return end
				debounceObj = other
				lastObj = other
				touchLabel.Text = " [DISENTUH]\n" .. describe(other)
				print("[DevHelper] disentuh:\n" .. describe(other))
				task.delay(0.3, function() debounceObj = nil end)
			end))
		end
	end
end

if player.Character then hookCharacter(player.Character) end

local esp = {} -- [obj] = {group, adornee, text, bb, hl, line, name}
local hlCount = 0

local espInfoLabel

local function updateEspCount()
	if not espInfoLabel then return end
	local n = 0
	for _ in pairs(esp) do n += 1 end
	espInfoLabel.Text = " ESP aktif: " .. n .. "\n Highlight dibatasi 30 (limit Roblox),\n sisanya tetap pakai label + jarak."
end

local function removeESP(obj)
	local e = esp[obj]
	if not e then return end
	if e.bb then e.bb:Destroy() end
	if e.line then e.line:Destroy() end
	if e.hl then
		e.hl:Destroy()
		hlCount -= 1
	end
	esp[obj] = nil
end

local function addESP(obj, group, color, tag)
	if esp[obj] then return end
	local adornee = obj
	if obj:IsA("Model") then
		adornee = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)
	end
	if not adornee or not adornee:IsA("BasePart") then return end

	local bb = make("BillboardGui", {
		Adornee = adornee,
		Size = UDim2.fromOffset(170, 36),
		AlwaysOnTop = true,
		StudsOffset = Vector3.new(0, 2, 0),
	}, gui)
	local txt = make("TextLabel", {
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		TextColor3 = color,
		TextStrokeTransparency = 0.3,
		Font = Enum.Font.GothamBold,
		TextSize = 12,
		Text = obj.Name,
	}, bb)

	local hl
	if hlCount < 30 then
		hl = make("Highlight", {
			Adornee = obj,
			FillColor = color,
			OutlineColor = color,
			FillTransparency = 0.7,
			DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
		}, obj)
		hlCount += 1
	end

	local line = make("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundColor3 = color,
		BorderSizePixel = 0,
		Visible = false,
		ZIndex = 1,
	}, gui)

	esp[obj] = {
		group = group,
		adornee = adornee,
		text = txt,
		bb = bb,
		hl = hl,
		line = line,
		name = obj.Name .. (tag and (" [" .. tag .. "]") or ""),
	}
end

local function clearGroup(group)
	for obj, e in pairs(esp) do
		if group == nil or e.group == group then removeESP(obj) end
	end
	updateEspCount()
end

RunService.RenderStepped:Connect(function()
	local hrp = getHRP()
	local cam = workspace.CurrentCamera
	local vs = cam and cam.ViewportSize or Vector2.new()
	local from = Vector2.new(vs.X / 2, vs.Y)

	for obj, e in pairs(esp) do
		if not obj.Parent or not e.adornee.Parent then
			removeESP(obj)
		else
			if hrp then
				local d = (e.adornee.Position - hrp.Position).Magnitude
				e.text.Text = showDist and string.format("%s\n[%d studs]", e.name, d) or e.name
			else
				e.text.Text = e.name
			end

			if tracersOn and cam then
				local sp, onScreen = cam:WorldToViewportPoint(e.adornee.Position)
				if onScreen then
					local to = Vector2.new(sp.X, sp.Y)
					local diff = to - from
					e.line.Visible = true
					e.line.Size = UDim2.fromOffset(diff.Magnitude, 1.5)
					e.line.Position = UDim2.fromOffset((from.X + to.X) / 2, (from.Y + to.Y) / 2)
					e.line.Rotation = math.deg(math.atan2(diff.Y, diff.X))
				else
					e.line.Visible = false
				end
			else
				e.line.Visible = false
			end
		end
	end
end)

-- ESP objek by nama / class
local function scanESP(mode)
	local key = keyBox.Text:lower()
	if key == "" then
		notify("isi kata kunci dulu")
		return
	end
	local color = COLORS[colorIdx][2]
	local n = 0
	for _, d in ipairs(workspace:GetDescendants()) do
		if (d:IsA("BasePart") or d:IsA("Model")) and not inCharacter(d) then
			local field = (mode == "name") and d.Name or d.ClassName
			if field:lower():find(key, 1, true) then
				addESP(d, "object", color)
				n += 1
				if n >= 150 then break end -- batas biar tidak lag
			end
		end
	end
	notify("ESP objek ditambahkan: " .. n)
	updateEspCount()
end

espInterBtn.MouseButton1Click:Connect(function()
	local color = COLORS[colorIdx][2]
	local n = 0
	for _, d in ipairs(workspace:GetDescendants()) do
		local tag
		if d:IsA("ProximityPrompt") then
			tag = "Prompt"
		elseif d:IsA("ClickDetector") then
			tag = "Click"
		elseif d:IsA("TouchTransmitter") then
			tag = "Touch"
		end
		if tag then
			local part = d.Parent
			if part and part:IsA("BasePart") and not inCharacter(part) then
				local m = part:FindFirstAncestorOfClass("Model")
				local isChar = m and m:FindFirstChildOfClass("Humanoid") ~= nil
				if not isChar then
					addESP(part, "object", color, tag)
					n += 1
					if n >= 150 then break end
				end
			end
		end
	end
	notify("objek interaktif: " .. n)
	updateEspCount()
end)

	notify("ESP objek dihapus")

local function tpTo(cf)
	local hrp = getHRP()
	if hrp then
		hrp.CFrame = cf + Vector3.new(0, 3, 0)
		notify("teleport berhasil")
	else
		notify("karakter tidak ditemukan")
	end
end

speedBtn.MouseButton1Click:Connect(function()
	local h, n = getHum(), tonumber(speedBox.Text)
	if h and n then
		origSpeed = origSpeed or h.WalkSpeed
		h.WalkSpeed = n
		notify("WalkSpeed = " .. n)
	else
		notify("angka tidak valid")
	end
end)

jumpBtn.MouseButton1Click:Connect(function()
	local h, n = getHum(), tonumber(jumpBox.Text)
	if h and n then
		origJump = origJump or h.JumpPower
		h.UseJumpPower = true
		h.JumpPower = n
		notify("JumpPower = " .. n)
	else
		notify("angka tidak valid")
	end
end)

resetBtn.MouseButton1Click:Connect(function()
	local h = getHum()

	if origJump then h.JumpPower = origJump end
	notify("speed/jump dikembalikan")
end)

	x, y, z = tonumber(x), tonumber(y), tonumber(z)
	if x and y and z then

		notify("format: x, y, z")
	end

local waypoints = {}

local function refreshWaypoints()
	clearList(wpList)
	for i, w in ipairs(waypoints) do
		local row = make("Frame", {
			Size = UDim2.new(1, -6, 0, 22),
			BackgroundTransparency = 1,
			LayoutOrder = i,
		}, wpList)
		local tp = make("TextButton", {
			Size = UDim2.new(1, -62, 1, 0),
			BackgroundColor3 = Color3.fromRGB(50, 50, 62),
			Text = " TP: " .. w.name,
			TextColor3 = Color3.new(1, 1, 1),
			Font = Enum.Font.Code,
			TextSize = 11,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextTruncate = Enum.TextTruncate.AtEnd,
			BorderSizePixel = 0,
		}, row)
		local cp = make("TextButton", {
			Size = UDim2.fromOffset(58, 22),
			Position = UDim2.new(1, -58, 0, 0),
			BackgroundColor3 = COL_OFF,
			Text = "Copy",
			TextColor3 = Color3.new(1, 1, 1),
			Font = Enum.Font.Code,
			TextSize = 11,
			BorderSizePixel = 0,
		}, row)
		tp.MouseButton1Click:Connect(function() tpTo(w.cf) end)
		cp.MouseButton1Click:Connect(function() copy(cfString(w.cf), "CFrame") end)
	end
end

wpSaveBtn.MouseButton1Click:Connect(function()
	local hrp = getHRP()
	if not hrp then
		notify("karakter tidak ditemukan")
		return
	end
	local name = wpBox.Text ~= "" and wpBox.Text or ("wp" .. (#waypoints + 1))
	table.insert(waypoints, { name = name, cf = hrp.CFrame })
	refreshWaypoints()
	notify("waypoint disimpan: " .. name)
end)
